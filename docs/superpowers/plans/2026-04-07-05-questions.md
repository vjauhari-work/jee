**Goal:** Implement question browsing endpoints: by year, by topic, list topics, list years, and get answer for a question.

**Architecture:** Questions handler queries MongoDB with filters on section/year/topic fields. The `correct_answer` field is excluded from responses (it's tagged `json:"-"` in the model) to prevent cheating. Answers are served separately via a dedicated endpoint.

package handlers

import (
	"context"
	"encoding/json"
	"net/http"
	"strconv"
	"time"

	"go.mongodb.org/mongo-driver/bson"
	"go.mongodb.org/mongo-driver/bson/primitive"
	"go.mongodb.org/mongo-driver/mongo/options"

	"github.com/vjauhari-work/jee/backend/db"
	"github.com/vjauhari-work/jee/backend/models"
)

type QuestionsHandler struct{}

func (h *QuestionsHandler) GetSections(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	sections := []map[string]string{
		{"id": models.SectionBoards11, "name": "Boards — Class 11", "description": "CBSE board exam preparation"},
		{"id": models.SectionBoards12, "name": "Boards — Class 12", "description": "CBSE board exam preparation"},
		{"id": models.SectionJEEMains, "name": "JEE Mains", "description": "Joint Entrance Examination"},
		{"id": models.SectionJEEAdvanced, "name": "JEE Advanced", "description": "Advanced level preparation"},
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(sections)
}

func (h *QuestionsHandler) GetByYear(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	section := r.URL.Query().Get("section")
	yearStr := r.URL.Query().Get("year")

	if section == "" || yearStr == "" {
		http.Error(w, `{"error":"section and year are required"}`, http.StatusBadRequest)
		return
	}

	year, err := strconv.Atoi(yearStr)
	if err != nil {
		http.Error(w, `{"error":"year must be a number"}`, http.StatusBadRequest)
		return
	}

	ctx, cancel := context.WithTimeout(r.Context(), 10*time.Second)
	defer cancel()

	filter := bson.M{"section": section, "year": year}
	cursor, err := db.Collection("questions").Find(ctx, filter, options.Find().SetSort(bson.D{{Key: "subject", Value: 1}}))
	if err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}
	defer cursor.Close(ctx)

	var questions []models.Question
	if err := cursor.All(ctx, &questions); err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(questions)
}

func (h *QuestionsHandler) GetByTopic(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	section := r.URL.Query().Get("section")
	subject := r.URL.Query().Get("subject")
	topic := r.URL.Query().Get("topic")

	if section == "" || subject == "" || topic == "" {
		http.Error(w, `{"error":"section, subject, and topic are required"}`, http.StatusBadRequest)
		return
	}

	ctx, cancel := context.WithTimeout(r.Context(), 10*time.Second)
	defer cancel()

	filter := bson.M{"section": section, "subject": subject, "topic": topic}
	cursor, err := db.Collection("questions").Find(ctx, filter, options.Find().SetSort(bson.D{{Key: "year", Value: -1}}))
	if err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}
	defer cursor.Close(ctx)

	var questions []models.Question
	if err := cursor.All(ctx, &questions); err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(questions)
}

func (h *QuestionsHandler) GetTopics(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	section := r.URL.Query().Get("section")
	subject := r.URL.Query().Get("subject")

	if section == "" || subject == "" {
		http.Error(w, `{"error":"section and subject are required"}`, http.StatusBadRequest)
		return

	ctx, cancel := context.WithTimeout(r.Context(), 10*time.Second)
	defer cancel()

	filter := bson.M{"section": section, "subject": subject}
	topics, err := db.Collection("questions").Distinct(ctx, "topic", filter)
	if err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(topics)
}

func (h *QuestionsHandler) GetYears(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	section := r.URL.Query().Get("section")
	if section == "" {

		return

	ctx, cancel := context.WithTimeout(r.Context(), 10*time.Second)
	defer cancel()

	years, err := db.Collection("questions").Distinct(ctx, "year", bson.M{"section": section})
	if err != nil {
		http.Error(w, `{"error":"internal server error"}`, http.StatusInternalServerError)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(years)
}

func (h *QuestionsHandler) GetAnswer(w http.ResponseWriter, r *http.Request) {
	if r.Method != http.MethodGet {
		http.Error(w, `{"error":"method not allowed"}`, http.StatusMethodNotAllowed)
		return
	}

	// Extract question ID from path: /api/questions/{id}/answer
	// We'll parse it manually since we're using net/http
	id := extractPathParam(r.URL.Path, "/api/questions/", "/answer")
	if id == "" {
		http.Error(w, `{"error":"question id is required"}`, http.StatusBadRequest)
		return
	}

	objID, err := primitive.ObjectIDFromHex(id)
	if err != nil {
		http.Error(w, `{"error":"invalid question id"}`, http.StatusBadRequest)
		return
	}

	ctx, cancel := context.WithTimeout(r.Context(), 5*time.Second)
	defer cancel()

	var answer models.Answer
	err = db.Collection("answers").FindOne(ctx, bson.M{"question_id": objID}).Decode(&answer)
	if err != nil {
		http.Error(w, `{"error":"answer not found"}`, http.StatusNotFound)
		return
	}

	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(answer)
}

// extractPathParam extracts a value between a prefix and suffix in a URL path.
// e.g., extractPathParam("/api/questions/abc123/answer", "/api/questions/", "/answer") returns "abc123"
func extractPathParam(path, prefix, suffix string) string {
	if len(path) <= len(prefix)+len(suffix) {
		return ""
	}
	start := len(prefix)
	end := len(path) - len(suffix)
	if start >= end || start < 0 || end > len(path) {
		return ""
	}
	return path[start:end]
}

package handlers

import "testing"

func TestExtractPathParam(t *testing.T) {
	tests := []struct {
		path, prefix, suffix, expected string
	}{
		{"/api/questions/abc123/answer", "/api/questions/", "/answer", "abc123"},
		{"/api/questions//answer", "/api/questions/", "/answer", ""},
		{"/api/quiz/sess456", "/api/quiz/", "", "sess456"},
		{"", "/api/", "/end", ""},
	}

	for _, tt := range tests {
		result := extractPathParam(tt.path, tt.prefix, tt.suffix)
		if result != tt.expected {
			t.Errorf("extractPathParam(%q, %q, %q) = %q, want %q",
				tt.path, tt.prefix, tt.suffix, result, tt.expected)
		}
	}
}
